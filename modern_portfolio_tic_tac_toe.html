Minimax Algorithm Implementation

Recursive Decision Tree logic for Unbeatable AI

How Unbeatable AI Works: The algorithm evaluates every possible future board state using recursion. Scores are assigned as +10 for AI win, -10 for Player win, and 0 for tie, subtracted by search depth to favor quicker wins. 

/* Minimax Engine Excerpt */
function minimax(board, depth, isMaximizing) {
    let score = evaluateBoard(board);

    if (score === 10) return score - depth;
    if (score === -10) return score + depth;
    if (!isMovesLeft(board)) return 0;

    if (isMaximizing) {
        let best = -Infinity;
        for (let i = 0; i < 9; i++) {
            if (board[i] === null) {
                board[i] = 'O'; // AI
                best = Math.max(best, minimax(board, depth + 1, false));
                board[i] = null;
            }
        }
        return best;
    } else {
        let best = Infinity;
        for (let i = 0; i < 9; i++) {
            if (board[i] === null) {
                board[i] = 'X'; // Human
                best = Math.min(best, minimax(board, depth + 1, true));
                board[i] = null;
            }
        }
        return best;
    }
}
